# Core audit.py runs on stdlib alone. These make it better:
requests            # more robust HTTP than urllib
beautifulsoup4      # HTML parsing for accessibility checks
dnspython           # SPF/DKIM/DMARC lookups without needing `dig`
certifi             # complete CA list so valid HTTPS sites verify (esp. on Windows)

# For the report step:
anthropic           # Claude API client

# For PDF export:
fpdf2               # writes the PDF
markdown            # report markdown -> HTML

# For the real-browser accessibility audit (uses installed Edge or Chrome):
playwright          # drives the browser
axe-playwright-python  # bundles axe-core
